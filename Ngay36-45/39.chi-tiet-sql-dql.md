## Chi tiết về SQL: DQL

Tiếp theo, chúng ta sử dụng cơ sở dữ liệu hệ thống chọn môn học của trường đã tạo trước đó để cùng tìm hiểu về thao tác truy vấn trong DML. Dù là lập trình viên hay chuyên viên phân tích dữ liệu, truy vấn đều vô cùng quan trọng, nó quyết định việc chúng ta có thể lấy được dữ liệu mình cần từ cơ sở dữ liệu quan hệ hay không. Bạn nên thực thi lại một lần nữa DDL tạo cơ sở dữ liệu/tạo bảng ở bài trước nữa và DML chèn dữ liệu ở bài trước, đảm bảo bảng và dữ liệu không có vấn đề gì rồi mới thực hiện các thao tác dưới đây.

```SQL
USE school;

-- Truy vấn tất cả thông tin của tất cả sinh viên
SELECT stu_id,
       stu_name,
       stu_sex,
       stu_birth,
       stu_addr,
       col_id
  FROM tb_student;

-- Truy vấn mã sinh viên, họ tên và quê quán của sinh viên (phép chiếu và đặt bí danh)
SELECT stu_id AS ma_sinh_vien,
       stu_name AS ho_ten,
       stu_addr AS que_quan
  FROM tb_student;

-- Truy vấn tên và số tín chỉ của tất cả môn học (phép chiếu và đặt bí danh)
SELECT cou_name AS ten_mon_hoc,
       cou_credit AS so_tin_chi
  FROM tb_course;

-- Truy vấn họ tên và ngày sinh của tất cả nữ sinh (lọc dữ liệu)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0;

-- Truy vấn họ tên và ngày sinh của các nữ sinh có quê quán là "Thành Đô, Tứ Xuyên" (Tứ Xuyên, Thành Đô) (lọc dữ liệu)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       AND stu_addr = 'Thành Đô, Tứ Xuyên';

-- Truy vấn các sinh viên có quê quán là "Thành Đô, Tứ Xuyên" hoặc giới tính là nữ (lọc dữ liệu)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       OR stu_addr = 'Thành Đô, Tứ Xuyên';

-- Truy vấn họ tên, giới tính và ngày sinh của tất cả sinh viên thế hệ 8x (sinh từ 1980 đến 1989) (lọc dữ liệu)
SELECT stu_name,
       stu_sex,
       stu_birth
  FROM tb_student
 WHERE '1980-1-1' <= stu_birth 
       AND stu_birth <= '1989-12-31';
       
SELECT stu_name,
       stu_sex,
       stu_birth
  FROM tb_student
 WHERE stu_birth BETWEEN '1980-1-1' AND '1989-12-31';

-- Truy vấn tên và số tín chỉ của các môn học có số tín chỉ lớn hơn 2 (lọc dữ liệu)
SELECT cou_name,
	   cou_credit
  FROM tb_course
 WHERE cou_credit > 2;

-- Truy vấn tên và số tín chỉ của các môn học có số tín chỉ là số lẻ (lọc dữ liệu)
SELECT cou_name,
	   cou_credit
  FROM tb_course
 WHERE cou_credit MOD 2 <> 0;

-- Truy vấn mã sinh viên của những người chọn môn học 1111 và có điểm thi trên 90 điểm (lọc dữ liệu)
SELECT stu_id
  FROM tb_record
 WHERE cou_id = 1111
       AND score > 90;

-- Truy vấn họ tên và giới tính của sinh viên tên là "杨过" (Dương Quá) (lọc dữ liệu)
SELECT stu_name AS ho_ten, 
       CASE stu_sex WHEN 1 THEN 'Nam' ELSE 'Nữ' END AS gioi_tinh
  FROM tb_student
 WHERE stu_name = '杨过';
 
SELECT stu_name AS ho_ten, 
       IF(stu_sex, 'Nam', 'Nữ') AS gioi_tinh
  FROM tb_student
 WHERE stu_name = '杨过';
    
-- Truy vấn họ tên và giới tính của sinh viên mang họ "杨" (khớp mờ)
-- Ký tự đại diện % khớp với 0 hoặc nhiều ký tự tùy ý
SELECT stu_name AS ho_ten, 
       CASE stu_sex WHEN 1 THEN 'Nam' ELSE 'Nữ' END AS gioi_tinh
  FROM tb_student
 WHERE stu_name LIKE '杨%';

-- Truy vấn họ tên và giới tính của sinh viên họ "杨" có tên gồm 2 chữ Hán (khớp mờ)
-- Ký tự đại diện _ khớp với đúng 1 ký tự
SELECT stu_name AS ho_ten, 
       CASE stu_sex WHEN 1 THEN 'Nam' ELSE 'Nữ' END AS gioi_tinh
  FROM tb_student
 WHERE stu_name LIKE '杨_';

-- Truy vấn họ tên và giới tính của sinh viên họ "杨" có tên gồm 3 chữ Hán (khớp mờ)
SELECT stu_name AS ho_ten, 
       CASE stu_sex WHEN 1 THEN 'Nam' ELSE 'Nữ' END AS gioi_tinh
  FROM tb_student
 WHERE stu_name LIKE '杨__';
 
-- Truy vấn mã sinh viên và họ tên của các sinh viên có chữ số cuối của mã sinh viên là 3 (khớp mờ)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_id LIKE '%3';

-- Truy vấn mã sinh viên và họ tên của các sinh viên có tên chứa chữ "不" hoặc chữ "嫣" (khớp mờ và phép hợp)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%不%'
       OR stu_name LIKE '%嫣%';
       
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%不%'
  UNION
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%嫣%';

-- Truy vấn mã sinh viên và họ tên của sinh viên họ "杨" hoặc họ "林" có tên 3 chữ (khớp mờ bằng biểu thức chính quy)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name REGEXP '[林杨][\\u4e00-\\u9fa5]{2}';

-- Truy vấn họ tên sinh viên chưa nhập quê quán (xử lý giá trị null/rỗng)
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) = ''
       OR stu_addr is null;
 
-- Truy vấn họ tên sinh viên đã nhập quê quán (xử lý giá trị null/rỗng)
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Truy vấn tất cả các ngày mà sinh viên đã chọn môn học (loại bỏ trùng lặp)
SELECT DISTINCT sel_date
  FROM tb_record;

-- Truy vấn quê quán của sinh viên (loại bỏ trùng lặp)
SELECT DISTINCT stu_addr
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Truy vấn họ tên và ngày sinh của nam sinh, sắp xếp theo tuổi từ lớn đến nhỏ (sắp xếp)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 1
  ORDER BY stu_birth ASC;
 
-- Bổ sung: Quy đổi ngày sinh ở trên thành tuổi (hàm ngày tháng, hàm số học)
SELECT stu_name AS ho_ten,
       FLOOR(DATEDIFF(CURDATE(), stu_birth) / 365) AS tuoi
  FROM tb_student
 WHERE stu_sex = 1
  ORDER BY tuoi DESC;

-- Truy vấn ngày sinh của sinh viên lớn tuổi nhất (hàm tổng hợp)
SELECT MIN(stu_birth)
  FROM tb_student;

-- Truy vấn ngày sinh của sinh viên nhỏ tuổi nhất (hàm tổng hợp)
SELECT MAX(stu_birth)
  FROM tb_student;

-- Truy vấn điểm thi cao nhất của môn học có mã 1111 (hàm tổng hợp)
SELECT MAX(score)
  FROM tb_record
 WHERE cou_id = 1111;

-- Truy vấn điểm thấp nhất, điểm cao nhất, điểm trung bình, độ lệch chuẩn, phương sai của sinh viên có mã 1001 (hàm tổng hợp)
SELECT MIN(score) AS diem_thap_nhat,
       MAX(score) AS diem_cao_nhat,
	   ROUND(AVG(score), 1) AS diem_trung_binh,
       STDDEV(score) AS do_lech_chuan,
       VARIANCE(score) AS phuong_sai
  FROM tb_record
 WHERE stu_id = 1001;

-- Truy vấn điểm trung bình của sinh viên có mã 1001, nếu có giá trị null thì tính null là 0 điểm (hàm tổng hợp)
SELECT ROUND(SUM(score) / COUNT(*), 1) AS diem_trung_binh
  FROM tb_record
 WHERE stu_id = 1001;

-- Truy vấn số lượng sinh viên nam và nữ (gom nhóm và hàm tổng hợp)
SELECT CASE stu_sex WHEN 1 THEN 'Nam' ELSE 'Nữ' END AS gioi_tinh,
       COUNT(*) AS so_luong
  FROM tb_student
 GROUP BY stu_sex;

-- Truy vấn số lượng sinh viên của từng khoa (gom nhóm và hàm tổng hợp)
SELECT col_id AS ma_khoa,
       COUNT(*) AS so_luong
  FROM tb_student
 GROUP BY col_id
  WITH ROLLUP;

-- Truy vấn số lượng sinh viên nam nữ của từng khoa (gom nhóm và hàm tổng hợp)
SELECT col_id AS ma_khoa,
       CASE stu_sex WHEN 1 THEN 'Nam' ELSE 'Nữ' END AS gioi_tinh,
       COUNT(*) AS so_luong
  FROM tb_student
 GROUP BY col_id, stu_sex;

-- Truy vấn mã sinh viên và điểm trung bình của từng sinh viên (gom nhóm và hàm tổng hợp)
SELECT stu_id AS ma_sinh_vien,
	   ROUND(AVG(score), 1) AS diem_trung_binh
  FROM tb_record
 GROUP BY stu_id;

-- Truy vấn mã sinh viên và điểm trung bình của những sinh viên có điểm trung bình từ 90 điểm trở lên (lọc dữ liệu sau khi gom nhóm)
SELECT stu_id AS ma_sinh_vien,
	   ROUND(AVG(score), 1) AS diem_trung_binh
  FROM tb_record
 GROUP BY stu_id
HAVING diem_trung_binh >= 90;

-- Truy vấn mã sinh viên và điểm trung bình của những sinh viên có điểm trung bình 3 môn 1111, 2222, 3333 từ 90 trở lên (lọc dữ liệu trước và sau gom nhóm)
SELECT stu_id AS ma_sinh_vien,
	   ROUND(AVG(score), 1) AS diem_trung_binh
  FROM tb_record
 WHERE cou_id in (1111, 2222, 3333)
 GROUP BY stu_id
HAVING diem_trung_binh >= 90
 ORDER BY diem_trung_binh ASC;

-- Truy vấn họ tên của sinh viên lớn tuổi nhất (truy vấn con)
SELECT stu_name
  FROM tb_student
 WHERE stu_birth = (SELECT MIN(stu_birth)
                      FROM tb_student);

-- Truy vấn họ tên của sinh viên đã chọn từ 2 môn học trở lên (truy vấn con và phép toán tập hợp)
SELECT stu_name
  FROM tb_student
 WHERE stu_id in (SELECT stu_id
				    FROM tb_record
				   GROUP BY stu_id
				  HAVING COUNT(*) > 2);

-- Truy vấn họ tên, ngày sinh và tên khoa trực thuộc của sinh viên (kết nối bảng)
SELECT stu_name,
       stu_birth,
       col_name
  FROM tb_student AS t1, tb_college AS t2
 WHERE t1.col_id = t2.col_id;
 
SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student INNER JOIN tb_college
       ON tb_student.col_id = tb_college.col_id;

SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student NATURAL JOIN tb_college;
  
SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student CROSS JOIN tb_college;

-- Truy vấn họ tên sinh viên, tên môn học và điểm thi (kết nối bảng)
SELECT stu_name,
       cou_name,
	   score
  FROM tb_student, tb_course, tb_record
 WHERE tb_student.stu_id = tb_record.stu_id
       AND tb_course.cou_id = tb_record.cou_id
       AND score is not null;

SELECT stu_name,
       cou_name,
	   score
  FROM tb_student 
       INNER JOIN tb_record
	       ON tb_student.stu_id = tb_record.stu_id
	   INNER JOIN tb_course
	       ON tb_course.cou_id = tb_record.cou_id
 WHERE score is not null;
 
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null;

-- Bổ sung: Lấy 5 dòng đầu tiên từ kết quả truy vấn trên (truy vấn phân trang)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5;

-- Bổ sung: Lấy dòng thứ 6-10 từ kết quả truy vấn trên (truy vấn phân trang)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5
OFFSET 5;

-- Bổ sung: Lấy dòng thứ 11-15 từ kết quả truy vấn trên (truy vấn phân trang)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 10, 5;

-- Truy vấn họ tên và điểm trung bình của sinh viên đã chọn môn (truy vấn con và kết nối bảng)
-- Error Code: 1248. Every derived table must have its own alias
SELECT stu_name,
	   avg_score
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            ROUND(AVG(score), 1) AS avg_score
                       FROM tb_record
                      GROUP BY stu_id) as tmp;

-- Truy vấn họ tên và số lượng môn học đã chọn của sinh viên (truy vấn con và kết nối bảng)
SELECT stu_name,
	   total
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            COUNT(*) AS total
                       FROM tb_record
                      GROUP BY stu_id) as tmp;

-- Truy vấn họ tên và số lượng môn học đã chọn của mỗi sinh viên (truy vấn con và left outer join)
SELECT stu_name AS ho_ten,
	   COALESCE(total, 0) AS so_mon_da_chon
  FROM tb_student AS t1
	   LEFT JOIN (SELECT stu_id,
			             COUNT(*) AS total
		            FROM tb_record
				   GROUP BY stu_id) AS t2
	       ON t1.stu_id = t2.stu_id;
```

Có một vài điểm cần được giải thích thêm:

1. Phiên bản hiện tại của MySQL không hỗ trợ phép nối ngoài toàn phần (full outer join). Ở trên, chúng ta thực hiện hiệu ứng nối ngoài toàn phần bằng cách dùng thao tác `union` để lấy hợp của kết quả phép nối ngoài bên trái (left outer join) và phép nối ngoài bên phải (right outer join). Các bạn có thể thông qua hình ảnh dưới đây để hiểu sâu hơn về các thao tác nối bảng:

   <img src="http://localhost/mypic/20211121135117.png" style="zoom:50%">

2. MySQL hỗ trợ nhiều loại toán tử khác nhau, bao gồm: toán tử số học (`+`, `-`, `*`, `/`, `%`), toán tử so sánh (`=`, `<>`, `<=>`, `<`, `<=`, `>`, `>=`, `BETWEEN...AND...`, `IN`, `IS NULL`, `IS NOT NULL`, `LIKE`, `RLIKE`, `REGEXP`), toán tử logic (`NOT`, `AND`, `OR`, `XOR`) và toán tử bit (`&`, `|`, `^`, `~`, `>>`, `<<`). Chúng ta có thể sử dụng các toán tử này trong DML để xử lý dữ liệu.

3. Khi truy vấn dữ liệu, có thể sử dụng các hàm trong câu lệnh `SELECT` và các mệnh đề của nó (như mệnh đề `WHERE`, `ORDER BY`, `HAVING`, v.v.). Các hàm này bao gồm hàm chuỗi, hàm số học, hàm thời gian và ngày tháng, hàm luồng điều khiển, v.v., như trong các bảng dưới đây.

   Các hàm chuỗi thường dùng:

   | Hàm | Chức năng |
   | --------------------------- | ----------------------------------------------------- |
   | `CONCAT`                    | Nối nhiều chuỗi thành một chuỗi |
   | `FORMAT`                    | Định dạng số thành chuỗi và chỉ định số chữ số thập phân cần giữ |
   | `FROM_BASE64` / `TO_BASE64` | Giải mã / mã hóa BASE64 |
   | `BIN` / `OCT` / `HEX`       | Chuyển đổi số thành chuỗi nhị phân / bát phân / thập lục phân |
   | `LOCATE`                    | Tìm vị trí của chuỗi con trong chuỗi |
   | `LEFT` / `RIGHT`            | Trả về số ký tự được chỉ định từ phía bên trái / bên phải của chuỗi |
   | `LENGTH` / `CHAR_LENGTH`    | Trả về độ dài chuỗi tính theo đơn vị byte / ký tự |
   | `LOWER` / `UPPER`           | Trả về dạng chữ thường / chữ hoa của chuỗi |
   | `LPAD` / `RPAD`             | Đệm thêm ký tự chỉ định vào bên trái / bên phải nếu chuỗi chưa đủ độ dài |
   | `LTRIM` / `RTRIM`           | Xóa khoảng trắng ở phía trước / phía sau chuỗi |
   | `ORD` / `CHAR`              | Trả về mã tương ứng với ký tự / trả về ký tự tương ứng với mã |
   | `STRCMP`                    | So sánh chuỗi, trả về -1, 0, 1 lần lượt biểu thị nhỏ hơn, bằng, lớn hơn |
   | `SUBSTRING`                 | Trả về chuỗi con trong phạm vi chỉ định của chuỗi |

   Các hàm số học thường dùng:

   | Hàm | Chức năng |
   | -------------------------------------------------------- | ---------------------------------- |
   | `ABS`                                                    | Trả về giá trị tuyệt đối của một số |
   | `CEILING` / `FLOOR`                                      | Trả về kết quả làm tròn lên / làm tròn xuống của một số |
   | `CONV`                                                   | Chuyển đổi một số từ hệ cơ số này sang hệ cơ số khác |
   | `CRC32`                                                  | Tính mã kiểm tra dư thừa tuần hoàn (CRC32) |
   | `EXP` / `LOG` / `LOG2` / `LOG10`                         | Tính hàm mũ / logarit |
   | `POW`                                                    | Tính lũy thừa |
   | `RAND`                                                   | Trả về số ngẫu nhiên trong khoảng [0, 1) |
   | `ROUND`                                                  | Trả về kết quả làm tròn của một số |
   | `SQRT`                                                   | Trả về căn bậc hai của một số |
   | `TRUNCATE`                                               | Cắt bỏ phần thập phân của số đến độ chính xác chỉ định |
   | `SIN` / `COS` / `TAN` / `COT` / `ASIN` / `ACOS` / `ATAN` | Các hàm lượng giác |

   Các hàm thời gian và ngày tháng thường dùng:

   | Hàm | Chức năng |
   | ----------------------------------------- | ------------------------------------- |
   | `CURDATE` / `CURTIME` / `NOW`             | Lấy ngày hiện tại / giờ hiện tại / ngày và giờ hiện tại |
   | `ADDDATE` / `SUBDATE`                     | Cộng / trừ hai biểu thức ngày tháng và trả về kết quả |
   | `DATE` / `TIME`                           | Lấy ngày / giờ từ chuỗi |
   | `YEAR` / `MONTH` / `DAY`                  | Lấy năm / tháng / ngày từ ngày tháng |
   | `HOUR` / `MINUTE` / `SECOND`              | Lấy giờ / phút / giây từ thời gian |
   | `DATEDIFF` / `TIMEDIFF` / `TIMESTAMPDIFF` | Trả về khoảng cách giữa hai biểu thức ngày giờ tính theo ngày / giờ |
   | `MAKEDATE` / `MAKETIME`                   | Tạo ra một giá trị ngày / giờ |

   Các hàm điều khiển luồng thường dùng:

   | Hàm | Chức năng |
   | -------- | ------------------------------------------------ |
   | `IF`     | Trả về các giá trị khác nhau tùy vào điều kiện đúng hay sai |
   | `IFNULL` | Nếu là NULL thì trả về giá trị chỉ định, ngược lại trả về chính nó |
   | `NULLIF` | Nếu hai biểu thức bằng nhau thì trả về NULL, ngược lại trả về giá trị biểu thức thứ nhất |

   Các hàm thường dùng khác:

   | Hàm | Chức năng |
   | -------------------------- | ----------------------------- |
   | `MD5` / `SHA1` / `SHA2`    | Trả về bản tóm lược băm (digest) tương ứng của chuỗi |
   | `CHARSET` / `COLLATION`    | Trả về bộ ký tự / quy tắc đối chiếu |
   | `USER` / `CURRENT_USER`    | Trả về người dùng hiện tại |
   | `DATABASE`                 | Trả về tên cơ sở dữ liệu hiện tại |
   | `VERSION`                  | Trả về phiên bản cơ sở dữ liệu hiện tại |
   | `FOUND_ROWS` / `ROW_COUNT` | Trả về số dòng tìm thấy / số dòng bị ảnh hưởng |
   | `LAST_INSERT_ID`           | Trả về giá trị khóa chính tự tăng cuối cùng |
   | `UUID` / `UUID_SHORT`      | Trả về định danh duy nhất toàn cầu |
